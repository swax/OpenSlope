cmake_minimum_required(VERSION 3.24)
project(OpenSlopeNative VERSION 0.1.0 LANGUAGES C CXX)

option(OPENSLOPE_BUILD_PLAYER "Build the Vulkan/OpenXR prototype" OFF)
option(OPENSLOPE_BUILD_TESTS "Build portable conformance tests" ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(cmake/dependencies.cmake)
openslope_dependency(json)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(json)
openslope_dependency(lodepng)
FetchContent_MakeAvailable(lodepng)
add_library(openslope_png STATIC "${lodepng_SOURCE_DIR}/lodepng.cpp")
set_target_properties(openslope_png PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(openslope_png SYSTEM PUBLIC "${lodepng_SOURCE_DIR}")
target_compile_definitions(openslope_png PUBLIC LODEPNG_NO_COMPILE_ENCODER LODEPNG_NO_COMPILE_DISK LODEPNG_NO_COMPILE_ANCILLARY_CHUNKS)

# Embed the tracked contract, rather than maintaining another table by hand.
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ../Contracts/ride-v1.json)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../Contracts/ride-v1.json" RIDE_CONTRACT_HEX HEX)
string(REGEX REPLACE "(..)" "0x\\1," RIDE_CONTRACT_BYTES "${RIDE_CONTRACT_HEX}")
configure_file(cmake/contract.hpp.in generated/contract.hpp @ONLY)
add_library(openslope_core src/terrain.cpp src/ride.cpp src/player.cpp src/disc.cpp src/import.cpp src/textures.cpp src/appearance.cpp src/props.cpp src/prop_import.cpp src/paths.cpp src/path_import.cpp src/replay.cpp)
target_include_directories(openslope_core PUBLIC include PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(openslope_core PRIVATE nlohmann_json::nlohmann_json openslope_png)
if(MSVC)
  target_compile_options(openslope_core PRIVATE /W4 /WX /permissive- /utf-8)
else()
  target_compile_options(openslope_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

add_executable(openslope_inspect src/inspect.cpp)
target_link_libraries(openslope_inspect PRIVATE openslope_core)
add_executable(openslope_import src/import_cli.cpp)
target_link_libraries(openslope_import PRIVATE openslope_core)
if(OPENSLOPE_BUILD_TESTS)
  enable_testing()
  add_executable(openslope_tests tests/core_tests.cpp)
  add_executable(openslope_appearance_tests tests/appearance_tests.cpp)
  add_executable(openslope_prop_tests tests/prop_tests.cpp)
  add_executable(openslope_gameplay_tests tests/player_tests.cpp)
  target_link_libraries(openslope_gameplay_tests PRIVATE openslope_core nlohmann_json::nlohmann_json)
  target_link_libraries(openslope_prop_tests PRIVATE openslope_core nlohmann_json::nlohmann_json)
  target_link_libraries(openslope_appearance_tests PRIVATE openslope_core)
  target_link_libraries(openslope_tests PRIVATE openslope_core nlohmann_json::nlohmann_json)
  add_test(NAME native_conformance COMMAND openslope_tests "${CMAKE_CURRENT_SOURCE_DIR}/../Contracts/carving-response-cases-v1.json")
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME native_iso_import COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/import_test.py" "$<TARGET_FILE:openslope_import>")
  add_test(NAME native_appearance COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/appearance_test.py" "$<TARGET_FILE:openslope_appearance_tests>")
  add_test(NAME native_props COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/appearance_test.py" "$<TARGET_FILE:openslope_prop_tests>")
  add_test(NAME native_prop_import COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/prop_import_test.py" "$<TARGET_FILE:openslope_import>")
  add_test(NAME native_gameplay COMMAND openslope_gameplay_tests)
  add_test(NAME native_path_import COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/path_import_test.py" "$<TARGET_FILE:openslope_import>")
endif()

if(OPENSLOPE_BUILD_PLAYER)
  include(cmake/player.cmake)
endif()
