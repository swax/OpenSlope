# Bundle the same GLBs as Slopesmith; separate translation units bound compiler memory.
set(character_catalog "${CMAKE_CURRENT_SOURCE_DIR}/../Slopesmith/src/core/characters/builtins.ts")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${character_catalog}")
file(STRINGS "${character_catalog}" CHARACTER_FILES REGEX "^    file: '")
file(STRINGS "${character_catalog}" CHARACTER_LABELS REGEX "^    label: '")
list(LENGTH CHARACTER_FILES CHARACTER_COUNT)
math(EXPR CHARACTER_LAST "${CHARACTER_COUNT} - 1")
set(CHARACTER_DECLARATIONS "")
set(CHARACTER_OPTIONS "")
set(CHARACTER_DISPATCH "")
set(CHARACTER_SOURCES "")
foreach(CHARACTER_INDEX RANGE 0 ${CHARACTER_LAST})
  list(GET CHARACTER_FILES ${CHARACTER_INDEX} character)
  string(REGEX REPLACE "^    file: '([^']+)'.*$" "\\1" character "${character}")
  list(GET CHARACTER_LABELS ${CHARACTER_INDEX} label)
  string(REGEX REPLACE "^    label: '([^']+)'.*$" "\\1" label "${label}")
  string(REPLACE "-rigged.glb" "" character_id "${character}")
  string(APPEND CHARACTER_DECLARATIONS "std::span<const unsigned char> character_${CHARACTER_INDEX}();\n")
  string(APPEND CHARACTER_OPTIONS "{\"builtin:${character_id}\", \"${label}\", {}},\n")
  string(APPEND CHARACTER_DISPATCH "if (id == \"builtin:${character_id}\") return bundled::character_${CHARACTER_INDEX}();\n")
  set(character_path "${CMAKE_CURRENT_SOURCE_DIR}/../Slopesmith/public/characters/${character}")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${character_path}")
  file(READ "${character_path}" CHARACTER_HEX HEX)
  string(REGEX REPLACE "(..)" "0x\\1," CHARACTER_BYTES "${CHARACTER_HEX}")
  configure_file(cmake/character.cpp.in generated/character_${CHARACTER_INDEX}.cpp @ONLY)
  list(APPEND CHARACTER_SOURCES "${CMAKE_CURRENT_BINARY_DIR}/generated/character_${CHARACTER_INDEX}.cpp")
endforeach()
configure_file(cmake/characters.cpp.in generated/characters.cpp @ONLY)
list(APPEND CHARACTER_SOURCES "${CMAKE_CURRENT_BINARY_DIR}/generated/characters.cpp")
