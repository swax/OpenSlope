cmake_minimum_required(VERSION 3.24)
project(OpenSlopeNative VERSION 0.1.0 LANGUAGES C CXX)

option(OPENSLOPE_BUILD_PLAYER "Build the Vulkan/OpenXR prototype" OFF)
option(OPENSLOPE_BUILD_TESTS "Build portable conformance tests" ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(ANDROID)
  # Keep installable debug APKs debuggable while optimizing their native game loop.
  add_compile_options("$<$<CONFIG:Debug>:-O2>")
endif()
include(cmake/dependencies.cmake)
openslope_dependency(json)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(json)
openslope_dependency(lodepng)
FetchContent_MakeAvailable(lodepng)
add_library(openslope_png STATIC "${lodepng_SOURCE_DIR}/lodepng.cpp")
set_target_properties(openslope_png PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(openslope_png SYSTEM PUBLIC "${lodepng_SOURCE_DIR}")
target_compile_definitions(openslope_png PUBLIC LODEPNG_NO_COMPILE_ENCODER LODEPNG_NO_COMPILE_DISK LODEPNG_NO_COMPILE_ANCILLARY_CHUNKS)

# Embed the tracked contract, rather than maintaining another table by hand.
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ../Contracts/ride-v1.json)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../Contracts/ride-v1.json" RIDE_CONTRACT_HEX HEX)
string(REGEX REPLACE "(..)" "0x\\1," RIDE_CONTRACT_BYTES "${RIDE_CONTRACT_HEX}")
configure_file(cmake/contract.hpp.in generated/contract.hpp @ONLY)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ../Contracts/external-sound-banks-v1.json)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../Contracts/external-sound-banks-v1.json" EXTERNAL_SOUND_HEX HEX)
string(REGEX REPLACE "(..)" "0x\\1," EXTERNAL_SOUND_BYTES "${EXTERNAL_SOUND_HEX}")
configure_file(cmake/external_sounds.hpp.in generated/external_sounds.hpp @ONLY)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS assets/stick-figure-rigged.glb)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/assets/stick-figure-rigged.glb" FIGURE_HEX HEX)
string(REGEX REPLACE "(..)" "0x\\1," FIGURE_BYTES "${FIGURE_HEX}")
configure_file(cmake/figure.hpp.in generated/figure.hpp @ONLY)
add_library(openslope_core src/course_run.cpp src/boarding.cpp src/avatar.cpp src/terrain.cpp src/world.cpp src/animation.cpp src/rails.cpp src/rail_import.cpp src/riding_effects.cpp src/visibility.cpp src/ride.cpp src/player.cpp src/disc.cpp src/import.cpp src/textures.cpp src/appearance.cpp src/props.cpp src/prop_import.cpp src/paths.cpp src/path_import.cpp src/replay.cpp src/collision_import.cpp src/collision_geometry.cpp src/collision_world.cpp src/audio_pcm.cpp src/audio_mixer.cpp src/audio_data.cpp src/music.cpp src/audio_program.cpp src/audio_scene.cpp src/audio_import.cpp)
target_include_directories(openslope_core PUBLIC include PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(openslope_core PRIVATE nlohmann_json::nlohmann_json openslope_png)
if(MSVC)
  target_compile_options(openslope_core PRIVATE /W4 /WX /permissive- /utf-8)
else()
  target_compile_options(openslope_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

add_executable(openslope_inspect src/inspect.cpp)
target_link_libraries(openslope_inspect PRIVATE openslope_core)
add_executable(openslope_import src/import_cli.cpp)
target_link_libraries(openslope_import PRIVATE openslope_core)
add_executable(openslope_audio src/audio_cli.cpp)
target_link_libraries(openslope_audio PRIVATE openslope_core nlohmann_json::nlohmann_json)
